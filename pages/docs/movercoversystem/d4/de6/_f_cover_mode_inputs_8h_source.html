54 virtual void Interpolate(const FMoverDataStructBase& From, const FMoverDataStructBase& To, float Pct) override
55 {
56 // Cast to our type - if this fails the collection is mixing incompatible types
57 const FCoverModeInputs* FromInputs = static_cast<const FCoverModeInputs*>(&From);
58 const FCoverModeInputs* ToInputs = static_cast<const FCoverModeInputs*>(&To);
59
60 if (!FromInputs || !ToInputs)
61 {
62 return;
63 }
64
65 // Interpolate continuous values
66 WallHitLocation = FMath::Lerp(FromInputs->WallHitLocation, ToInputs->WallHitLocation, Pct);
67 WallHitNormal = FMath::Lerp(FromInputs->WallHitNormal, ToInputs->WallHitNormal, Pct).GetSafeNormal();
68 WallHitNormalRight = FMath::Lerp(FromInputs->WallHitNormalRight, ToInputs->WallHitNormalRight, Pct).GetSafeNormal();
69 CornerPoint = FMath::Lerp(FromInputs->CornerPoint, ToInputs->CornerPoint, Pct);
70 CornerWallNormal = FMath::Lerp(FromInputs->CornerWallNormal, ToInputs->CornerWallNormal, Pct).GetSafeNormal();
71 CornerWallRight = FMath::Lerp(FromInputs->CornerWallRight, ToInputs->CornerWallRight, Pct).GetSafeNormal();
72 SideWallNormal = FMath::Lerp(FromInputs->SideWallNormal, ToInputs->SideWallNormal, Pct).GetSafeNormal();
73 SideWallRight = FMath::Lerp(FromInputs->SideWallRight, ToInputs->SideWallRight, Pct).GetSafeNormal();
74 BlockedShimmyDirection = FMath::Lerp(FromInputs->BlockedShimmyDirection, ToInputs->BlockedShimmyDirection, Pct);
75
76 // Snap boolean/discrete values to whichever frame is dominant (past 50% use To values)
77 bShouldBeInCoverMode = Pct < 0.5f ? FromInputs->bShouldBeInCoverMode : ToInputs->bShouldBeInCoverMode;
78 bHasValidWallAhead = Pct < 0.5f ? FromInputs->bHasValidWallAhead : ToInputs->bHasValidWallAhead;
79 bWantsToAim = Pct < 0.5f ? FromInputs->bWantsToAim : ToInputs->bWantsToAim;
80 bIsAtCorner = Pct < 0.5f ? FromInputs->bIsAtCorner : ToInputs->bIsAtCorner;
81 bWantsToGoAroundCorner = Pct < 0.5f ? FromInputs->bWantsToGoAroundCorner : ToInputs->bWantsToGoAroundCorner;
82 bSideWallDetected = Pct < 0.5f ? FromInputs->bSideWallDetected : ToInputs->bSideWallDetected;
83 bEdgeFloorExists = Pct < 0.5f ? FromInputs->bEdgeFloorExists : ToInputs->bEdgeFloorExists;
