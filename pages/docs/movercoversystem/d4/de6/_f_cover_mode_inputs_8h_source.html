30 virtual bool NetSerialize(FArchive& Ar, UPackageMap* Map, bool& bOutSuccess) override
31 {
32 bool bSuccess = Super::NetSerialize(Ar, Map, bOutSuccess);
33
34 // Booleans - 1 bit each via SerializeBits
35 uint32 Flags = 0;
36 if (Ar.IsSaving())
37 {
38 Flags =
39 ((uint32)bShouldBeInCoverMode << 0) |
40 ((uint32)bHasValidWallAhead << 1) |
41 ((uint32)bWantsToAim << 2) |
42 ((uint32)bIsAtCorner << 3) |
43 ((uint32)bWantsToGoAroundCorner << 4) |
44 ((uint32)bSideWallDetected << 5) |
45 ((uint32)bEdgeFloorExists << 6) |
46 ((uint32)bGapDetected << 7) |
47 ((uint32)bWantsGapMovement << 8) |
48 ((uint32)bIsGapSprint << 9) |
49 ((uint32)bGapArrived << 10);
50 }
51 Ar.SerializeBits(&Flags, 11);
52 if (Ar.IsLoading())
53 {
54 bShouldBeInCoverMode = (Flags >> 0) & 1;
55 bHasValidWallAhead = (Flags >> 1) & 1;
56 bWantsToAim = (Flags >> 2) & 1;
57 bIsAtCorner = (Flags >> 3) & 1;
58 bWantsToGoAroundCorner = (Flags >> 4) & 1;
59 bSideWallDetected = (Flags >> 5) & 1;
60 bEdgeFloorExists = (Flags >> 6) & 1;
61 bGapDetected = (Flags >> 7) & 1;
62 bWantsGapMovement = (Flags >> 8) & 1;
63 bIsGapSprint = (Flags >> 9) & 1;
64 bGapArrived = (Flags >> 10) & 1;
65 }
66
67 // BlockedShimmyDirection is always -1, 0, or 1 - encode as 2 bits
68 uint8 ShimmyDir = 0;
69 if (Ar.IsSaving()) ShimmyDir = (uint8)(BlockedShimmyDirection + 1.0f);
70 Ar.SerializeBits(&ShimmyDir, 2);
71 if (Ar.IsLoading()) BlockedShimmyDirection = (float)ShimmyDir - 1.0f;
72
73 // Quantized normals -- 16 bits per component (sufficient for unit vectors)
74 auto SerializeNormal = [&](FVector& V)
75 {
76 uint16 X = 0, Y = 0, Z = 0;
77 if (Ar.IsSaving())
78 {
79 X = (uint16)FMath::Clamp(FMath::RoundToInt((V.X + 1.0f) * 32767.0f), 0, 65535);
80 Y = (uint16)FMath::Clamp(FMath::RoundToInt((V.Y + 1.0f) * 32767.0f), 0, 65535);
81 Z = (uint16)FMath::Clamp(FMath::RoundToInt((V.Z + 1.0f) * 32767.0f), 0, 65535);
82 }
83 Ar << X; Ar << Y; Ar << Z;
84 if (Ar.IsLoading())
85 {
86 V.X = X / 32767.0f - 1.0f;
87 V.Y = Y / 32767.0f - 1.0f;
88 V.Z = Z / 32767.0f - 1.0f;
89 V = V.GetSafeNormal();
90 }
91 };
92
93 // Quantized world location - 32 bits per component at 1cm precision
94 auto SerializeLocation = [&](FVector& V)
95 {
96 int32 X = 0, Y = 0, Z = 0;
97 if (Ar.IsSaving()) { X = (int32)V.X; Y = (int32)V.Y; Z = (int32)V.Z; }
98 Ar << X; Ar << Y; Ar << Z;
99 if (Ar.IsLoading()) { V.X = (float)X; V.Y = (float)Y; V.Z = (float)Z; }
100 };
101
102 SerializeLocation(WallHitLocation);
103 SerializeNormal(WallHitNormal);
104 SerializeNormal(WallHitNormalRight);
105 SerializeNormal(SideWallNormal);
106 SerializeNormal(SideWallRight);
107 SerializeLocation(CornerPoint);
108 SerializeNormal(CornerWallNormal);
109 SerializeNormal(CornerWallRight);
110 SerializeLocation(GapWallLocation);
111 SerializeNormal(GapWallNormal);
112 SerializeNormal(GapWallRight);
113
114 // GapDistance - 16 bits at 1cm precision, max 65535cm
115 uint16 GapDistQ = 0;
116 if (Ar.IsSaving()) GapDistQ = (uint16)FMath::Clamp(FMath::RoundToInt(GapDistance), 0, 65535);
117 Ar << GapDistQ;
118 if (Ar.IsLoading()) GapDistance = (float)GapDistQ;
119
120 bOutSuccess = !Ar.IsError();
121 return bSuccess;
122 }

124 virtual void Interpolate(const FMoverDataStructBase& From, const FMoverDataStructBase& To, float Pct) override
125 {
126 // Cast to our type - if this fails the collection is mixing incompatible types
127 const FCoverModeInputs* FromInputs = static_cast<const FCoverModeInputs*>(&From);
128 const FCoverModeInputs* ToInputs = static_cast<const FCoverModeInputs*>(&To);
129
130 if (!FromInputs || !ToInputs)
131 {
132 return;
133 }
134
135 // Interpolate continuous values
136 WallHitLocation = FMath::Lerp(FromInputs->WallHitLocation, ToInputs->WallHitLocation, Pct);
137 WallHitNormal = FMath::Lerp(FromInputs->WallHitNormal, ToInputs->WallHitNormal, Pct).GetSafeNormal();
138 WallHitNormalRight = FMath::Lerp(FromInputs->WallHitNormalRight, ToInputs->WallHitNormalRight, Pct).GetSafeNormal();
139 CornerPoint = FMath::Lerp(FromInputs->CornerPoint, ToInputs->CornerPoint, Pct);
140 CornerWallNormal = FMath::Lerp(FromInputs->CornerWallNormal, ToInputs->CornerWallNormal, Pct).GetSafeNormal();
141 CornerWallRight = FMath::Lerp(FromInputs->CornerWallRight, ToInputs->CornerWallRight, Pct).GetSafeNormal();
142 SideWallNormal = FMath::Lerp(FromInputs->SideWallNormal, ToInputs->SideWallNormal, Pct).GetSafeNormal();
143 SideWallRight = FMath::Lerp(FromInputs->SideWallRight, ToInputs->SideWallRight, Pct).GetSafeNormal();
144 BlockedShimmyDirection = FMath::Lerp(FromInputs->BlockedShimmyDirection, ToInputs->BlockedShimmyDirection, Pct);
145 GapWallLocation = FMath::Lerp(FromInputs->GapWallLocation, ToInputs->GapWallLocation, Pct);
146 GapWallNormal = FMath::Lerp(FromInputs->GapWallNormal, ToInputs->GapWallNormal, Pct).GetSafeNormal();
147 GapWallRight = FMath::Lerp(FromInputs->GapWallRight, ToInputs->GapWallRight, Pct).GetSafeNormal();
148 GapDistance = FMath::Lerp(FromInputs->GapDistance, ToInputs->GapDistance, Pct);
149
150 // Snap boolean/discrete values to whichever frame is dominant (past 50% use To values)
151 bShouldBeInCoverMode = Pct < 0.5f ? FromInputs->bShouldBeInCoverMode : ToInputs->bShouldBeInCoverMode;
152 bHasValidWallAhead = Pct < 0.5f ? FromInputs->bHasValidWallAhead : ToInputs->bHasValidWallAhead;
153 bWantsToAim = Pct < 0.5f ? FromInputs->bWantsToAim : ToInputs->bWantsToAim;
154 bIsAtCorner = Pct < 0.5f ? FromInputs->bIsAtCorner : ToInputs->bIsAtCorner;
155 bWantsToGoAroundCorner = Pct < 0.5f ? FromInputs->bWantsToGoAroundCorner : ToInputs->bWantsToGoAroundCorner;
156 bSideWallDetected = Pct < 0.5f ? FromInputs->bSideWallDetected : ToInputs->bSideWallDetected;
157 bEdgeFloorExists = Pct < 0.5f ? FromInputs->bEdgeFloorExists : ToInputs->bEdgeFloorExists;
158 bGapDetected = Pct < 0.5f ? FromInputs->bGapDetected : ToInputs->bGapDetected;
159 bWantsGapMovement = Pct < 0.5f ? FromInputs->bWantsGapMovement : ToInputs->bWantsGapMovement;
160 bIsGapSprint = Pct < 0.5f ? FromInputs->bIsGapSprint : ToInputs->bIsGapSprint;
161 bGapArrived = Pct < 0.5f ? FromInputs->bGapArrived : ToInputs->bGapArrived;
162 }
